import { CoreSyncClient } from '@coresync/sdk';

// 1. تهيئة العميل
const client = new CoreSyncClient({ apiKey: 'sr_live_2026_eu_prod_key' });

// 2. إصدار فاتورة متوافقة مع Belastingdienst
const invoice = await client.generateInvoice({
  clientName: 'Rotterdam Port Logistics B.V.',
  clientVat: 'NL854721963B01',
  amountNet: 2400.0,
  btwRate: 0.21
});
console.log('Invoice Created:', invoice.data.invoice_id);

// 3. إشعار WhatsApp فوري لتأكيد التسوية البنكية
const notify = await client.triggerWhatsAppNotification({
  managerPhone: '+31687654321',
  invoiceId: invoice.data.invoice_id,
  amount: 2904.0,
  ledgerRef: invoice.data.ledger_reference
});
console.log('WhatsApp Status:', notify.message);
